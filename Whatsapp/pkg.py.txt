# PYTHON PACKAGES: NumPy, Pandas, Matplotlib, Scikit-Learn
# ===========================================
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler

# ---------- NumPy ----------
arr = np.array([10, 20, 30, 40])
print("NumPy Array:", arr)
print("Mean:", np.mean(arr))
print()

# ---------- Pandas ----------
data = {
    "Name": ["Amit", "Riya", "Karan", "Sara"],
    "Age": [25, 28, 22, 24],
    "Marks": [85, 90, 78, 92]
}
df = pd.DataFrame(data)
print("Pandas DataFrame:\n", df, "\n")

# ---------- Matplotlib ----------
plt.bar(df["Name"], df["Marks"], color='skyblue')
plt.title("Student Marks Visualization")
plt.xlabel("Name")
plt.ylabel("Marks")
plt.show()

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler

# ---------- NumPy ----------
arr = np.array([10, 20, 30, 40])
print("NumPy Array:", arr)
print("Mean:", np.mean(arr))
print()

# ---------- Pandas ----------
data = {
    "Name": ["Amit", "Riya", "Karan", "Sara"],
    "Age": [25, 28, 22, 24],
    "Marks": [85, 90, 78, 92]
}
df = pd.DataFrame(data)
print("Pandas DataFrame:\n", df, "\n")

# ---------- Matplotlib ----------
plt.bar(df["Name"], df["Marks"], color='skyblue')
plt.title("Student Marks Visualization")
plt.xlabel("Name")
plt.ylabel("Marks")
plt.show()

# ===========================================
# IMPORTING DATASETS (FILE FORMATS)
# ===========================================
# Simulate saving and reading CSV
df.to_csv("students.csv", index=False)
print("CSV File Saved. Reading it back...\n")
new_df = pd.read_csv("students.csv")
print(new_df, "\n")

# ===========================================
# DATA PREPROCESSING (with Scikit-Learn)
# ===========================================
# Example numeric data for scaling
marks = df[["Marks"]].values
scaler = MinMaxScaler(feature_range=(0, 1))
scaled_marks = scaler.fit_transform(marks)

df["Normalized_Marks"] = scaled_marks
print("After Normalization:\n", df)
